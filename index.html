// 全域變數追蹤當前選擇的期程 let currentViewPeriod = '期中'; // --- 1. 學生登錄與輸入框 (記憶功能) --- document.getElementById('daily-form').addEventListener('submit', function(e) { e.preventDefault(); // 取得資料 const period = document.getElementById('daily-period').value; const subcat = document.getElementById('daily-subcat').value; const subject = document.getElementById('daily-subject-val').value; // TODO: 儲存資料至系統數據庫 (包含 period) alert(`【${period}】平時成績已成功提交！`); // 記憶保留：不清空 daily-period, subcat, subject 等下拉選單。 // 僅清空分數與課別勾選框 document.getElementById('daily-score-val').value = ''; document.querySelectorAll('#lesson-checkboxes input[type="checkbox"]').forEach(cb => cb.checked = false); // 重新渲染當下選擇的期程畫面 renderDailySection(currentViewPeriod); }); // --- 2. 處理定期評量登錄 (不跳轉欄位) --- // 攔截定期評量 input 的 Enter 鍵行為，防止預設的 form 提交或自動跳轉下一個 input document.querySelectorAll('.exam-score-input').forEach(input => { input.addEventListener('keydown', function(e) { if (e.key === 'Enter' || e.keyCode === 13) { e.preventDefault(); // 僅觸發儲存邏輯，焦點 (focus) 依然保留在當前欄位，方便老師確認或修改 saveExamScore(this.dataset.studentId, this.dataset.subject, this.value); // 顯示已儲存提示 (例如旁邊亮綠燈)，不改變 focus() } }); }); // --- 3. 學生端展示頁面 (切換 Tab 與渲染四大區塊) --- function switchPeriodTab(period) { currentViewPeriod = period; // 更新 Tab 樣式 document.querySelectorAll('.tab-btn').forEach(btn => { btn.classList.remove('text-blue-600', 'border-b-2', 'border-blue-600'); btn.classList.add('text-slate-500'); }); const activeTab = period === '期中' ? document.getElementById('tab-midterm') : document.getElementById('tab-final'); activeTab.classList.remove('text-slate-500'); activeTab.classList.add('text-blue-600', 'border-b-2', 'border-blue-600'); renderDailySection(period); } function renderDailySection(period) { const student = getCurrentStudent(); // 取得該學生「特定期程」的所有紀錄 const allRecords = getStudentDailyRecords(student); const periodRecords = allRecords.filter(r => r.period === period); // 渲染四大區塊 (沿用 v7.0 邏輯，但資料源改為 periodRecords) // 1. 平時評量 (五科 + 總分) // 2. 雲端速測 // 3. 小考分數 // 4. 作業分數 // 示意呼叫，具體渲染 DOM 結構與 v7.0 相同 renderRegularBlock(periodRecords); renderListBlock('block-cloud', periodRecords.filter(r => r.subcat === '雲端速測')); renderListBlock('block-quiz', periodRecords.filter(r => r.subcat === '小考分數')); renderListBlock('block-homework', periodRecords.filter(r => r.subcat === '作業分數')); } // --- 4. 老師端結算 Logic 與 學生端定期評量統計 --- const GRADE_CONVERT = { '甲上上': 95, '甲上': 90, '甲': 85, '甲下': 80, '甲下下': 75 }; // 老師端：依據期程結算平時成績 function calculateDailySummaryByPeriod(targetPeriod) { const students = getAllStudents(); let summaryResults = []; students.forEach(student => { const records = getStudentDailyRecords(student.name).filter(r => r.period === targetPeriod); // 雲端速測與小考平均邏輯 const cloudAvg = calcAverage(records.filter(r => r.subcat === '雲端速測')); const quizAvg = calcAverage(records.filter(r => r.subcat === '小考分數')); // 作業平均 (含等第轉換) const hwRecords = records.filter(r => r.subcat === '作業分數'); let hwAvg = '無紀錄'; if (hwRecords.length > 0) { const totalHw = hwRecords.reduce((sum, r) => sum + (r.score ? Number(r.score) : (GRADE_CONVERT[r.grade] || 0)), 0); hwAvg = (totalHw / hwRecords.length).toFixed(1); } summaryResults.push({ studentName: student.name, cloudAvg, quizAvg, hwAvg }); }); renderTeacherSummaryTable(summaryResults); } // 學生/家長端：顯示定期評量總分、排名與班級各科平均 function renderStudentExamStats() { const student = getCurrentStudent(); const students = getAllStudents(); let classSubjectTotals = { '國語': 0, '英語': 0, '數學': 0, '自然': 0, '社會': 0 }; let examData = []; // 計算全班資料以取得排名與平均 students.forEach(s => { const examRecords = getStudentExamRecords(s.name); let totalScore = 0; examRecords.forEach(r => { const score = Number(r.score || 0); totalScore += score; if (classSubjectTotals[r.subject] !== undefined) { classSubjectTotals[r.subject] += score; } }); examData.push({ studentName: s.name, totalScore }); }); // 處理排名 (處理同分同名次) examData.sort((a, b) => b.totalScore - a.totalScore); examData.forEach((item, index) => { if (index > 0 && item.totalScore === examData[index - 1].totalScore) { item.rank = examData[index - 1].rank; } else { item.rank = index + 1; } }); // 取得當前學生資料 const myData = examData.find(d => d.studentName === student.name); const studentCount = students.length; // 產出 HTML let html = `
我的五科總分
${myData.totalScore} 分
班級排名
第 ${myData.rank} 名

班級各科平均

`; Object.keys(classSubjectTotals).forEach(sub => { const avg = (classSubjectTotals[sub] / studentCount).toFixed(1); html += `
${sub} ${avg} 分
`; }); html += `
`; document.getElementById('exam-stats-container').innerHTML = html; } // 輔助計算平均 function calcAverage(records) { if (records.length === 0) return '無紀錄'; const total = records.reduce((sum, r) => sum + Number(r.score), 0); return (total / records.length).toFixed(1); }
